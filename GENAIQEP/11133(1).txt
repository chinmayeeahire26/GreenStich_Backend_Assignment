// SolutionSetupController.java - saveSolutionSettings() method snippet
public void saveSolutionSettings(Solution solution, String jiraUrl, String jiraToken, String jiraUsername) {
    // Enforce single Jira URL per solution
    if (solution.getJiraUrl() != null && !solution.getJiraUrl().isEmpty()) {
        if (!solution.getJiraUrl().equals(jiraUrl)) {
            // Reject change to Jira URL - keep existing URL
            // Optionally log or notify user about restriction
            // Do not update Jira URL
        }
    } else {
        // Jira URL not set yet - allow setting it
        solution.setJiraUrl(jiraUrl);
    }
    // Allow updating token and username regardless
    solution.setJiraToken(jiraToken);
    solution.setJiraUsername(jiraUsername);
    solutionRepository.save(solution);
}
