1) Description on code fix: The defect DF-25 states that when a defect is assigned/reassigned to another user, no notification email is received and SMTP logs show no outbound trigger event. The RCA points to two primary causes: (1) configuration where the notification trigger/configuration isn’t firing, and (2) code issue where the assignment-change event is not being emitted when reassignment happens. In the provided code, the only implementation is Selenium UI automation that logs in, selects a portfolio/project, and logs out. It never performs the core action (defect reassignment + save/update) and never validates that reassignment was persisted. As a result, this code cannot reproduce or validate DF-25 and can misleadingly align with “no outbound trigger event” simply because the trigger action never executes (or fails due to brittle locators / missing waits). The fix therefore updates the automation to (a) actually navigate to a defect, change the assignee, and click Save/Update—so the backend assignment-change path (which should emit the notification trigger per RCA) is executed; (b) add explicit waits and more stable locators so the reassignment action is reliably performed (avoiding false “no trigger” outcomes caused by race conditions); (c) fail fast instead of swallowing exceptions so incomplete workflows don’t produce misleading test results; and (d) externalize credentials/environment parameters so tests run against the correct tenant where notification configuration (RCA “Configuration”) is expected to be enabled.

package qConnect;

/
 * Fix intent:
 * - Actually perform defect reassignment + save to exercise notification trigger path (DF-25).
 * - Replace brittle absolute XPaths with stable locators where possible.
 * - Add explicit waits to avoid race conditions (Angular/Material UI).
 * - Fail fast with meaningful errors instead of swallowing exceptions.
 * - Externalize credentials/config to avoid environment drift (SMTP/notification config differs by tenant).
 */
public class DuoLoginAutomation {

	public static void loginToDuo() {
		// CHANGE: externalize driver path & credentials (avoid hard-coding)
		String chromeDriverPath = System.getProperty("webdriver.chrome.driver", "C:chromedriver-win64chromedriver.exe");
		System.setProperty("webdriver.chrome.driver", chromeDriverPath);

		String baseUrl = System.getProperty("qconnect.baseUrl", "https://qconnect.mpoweredhealth.com/login");
		String username = System.getProperty("qconnect.username"); // e.g. -Dqconnect.username=user@x.com
		String password = System.getProperty("qconnect.password"); // e.g. -Dqconnect.password=...

		if (username == null || password == null) {
			throw new IllegalArgumentException("Missing credentials: supply -Dqconnect.username and -Dqconnect.password");
		}

		ChromeOptions options = new ChromeOptions();
		options.addArguments("--start-maximized", "--disable-notifications", "--disable-geolocation", "--incognito");

		WebDriver driver = new ChromeDriver(options);
		WebDriverWait wait = new WebDriverWait(driver, Duration.ofSeconds(20)); // CHANGE: explicit waits

		try {
			driver.manage().timeouts().pageLoadTimeout(Duration.ofSeconds(30));
			driver.get(baseUrl);

			// --- Login ---
			wait.until(ExpectedConditions.visibilityOfElementLocated(By.id("email"))).sendKeys(username); // CHANGE: stable locator
			wait.until(ExpectedConditions.visibilityOfElementLocated(By.id("password"))).sendKeys(password);
			wait.until(ExpectedConditions.elementToBeClickable(By.xpath("//button[normalize-space()='Log In']"))).click();

			// --- Select portfolio / project (keep as-is but with waits) ---
			wait.until(ExpectedConditions.elementToBeClickable(By.cssSelector("mat-select#mat-select-0"))).click(); // CHANGE: css selector
			wait.until(ExpectedConditions.elementToBeClickable(By.xpath("//span[normalize-space()='Qeats - Sample Portfolio']"))).click();
			wait.until(ExpectedConditions.elementToBeClickable(By.xpath("//button[normalize-space()='Submit']"))).click();

			// NOTE: The original code navigates to module management via an absolute XPath.
			// CHANGE: wait for and click a more stable element if available; fallback kept minimal.
			// If the app has a menu link like "Testing" or "Defects", prefer that.
			// Example placeholder (update to your app’s real selector):
			// wait.until(ExpectedConditions.elementToBeClickable(By.xpath("//a[.='Testing']"))).click();

			// --- NEW: Navigate to Defects and reassign a defect ---
			// IMPORTANT: Actual selectors for "Defects", defect row, assignee field, and save button
			// are not present in the provided snippet. The following code is a template that must be
			// aligned to the real DOM.
			//
			// CHANGE: add the missing core action under test (assignment change + save)
			wait.until(ExpectedConditions.elementToBeClickable(By.xpath("//span[normalize-space()='Qeats - Sample Project']"))).click();
			wait.until(ExpectedConditions.elementToBeClickable(By.xpath("//button[@type='submit']"))).click();

			// ---- BEGIN DF-25 workflow (template selectors) ----
			// 1) Go to defects list
			wait.until(ExpectedConditions.elementToBeClickable(By.xpath("//a[contains(.,'Defects') or contains(.,'Bugs')]"))).click();

			// 2) Open a specific defect (use a known defect ID provided via system property)
			String defectId = System.getProperty("qconnect.defectId"); // e.g. DF-25 or internal numeric id
			if (defectId == null) {
				throw new IllegalArgumentException("Missing defect id: supply -Dqconnect.defectId=<id-to-reassign>");
			}
			wait.until(ExpectedConditions.elementToBeClickable(By.xpath("//a[contains(.,'" + defectId + "')]"))).click();

			// 3) Change assignee (example: Angular Material mat-select)
			String newAssignee = System.getProperty("qconnect.newAssignee"); // e.g. "naman.kumar@..."
			if (newAssignee == null) {
				throw new IllegalArgumentException("Missing new assignee: supply -Dqconnect.newAssignee=<user-display-or-email>");
			}

			WebElement assigneeSelect = wait.until(
					ExpectedConditions.elementToBeClickable(By.xpath("//mat-label[normalize-space()='Assignee']/ancestor::mat-form-field//mat-select"))
			);
			assigneeSelect.click();

			wait.until(ExpectedConditions.elementToBeClickable(By.xpath("//mat-option//span[contains(normalize-space(),'" + newAssignee + "')]")))
					.click();

			// 4) Save/Update defect (this is what should trigger backend event -> SMTP outbound)
			wait.until(ExpectedConditions.elementToBeClickable(By.xpath("//button[normalize-space()='Save' or normalize-space()='Update']"))).click();

			// 5) Assert assignee changed (prevents false "no SMTP event" due to no persistence)
			wait.until(ExpectedConditions.visibilityOfElementLocated(
					By.xpath("//mat-label[normalize-space()='Assignee']/ancestor::mat-form-field//*[contains(.,'" + newAssignee + "')]")
			));
			// ---- END DF-25 workflow ----

			// --- Logout ---
			wait.until(ExpectedConditions.elementToBeClickable(By.xpath("//mat-icon[normalize-space()='person']"))).click();
			wait.until(ExpectedConditions.elementToBeClickable(By.xpath("//button[normalize-space()='Logout']"))).click();

		} finally {
			driver.quit();
		}
