using OpenQA.Selenium.Support.UI; // Added for explicit waits

public void Setup()
{
    driver = new ChromeDriver();
    driver.Navigate().GoToUrl("https://example.com/login");

    // Added explicit wait to ensure page and username field are loaded
    WebDriverWait wait = new WebDriverWait(driver, TimeSpan.FromSeconds(10));
    wait.Until(SeleniumExtras.WaitHelpers.ExpectedConditions.ElementIsVisible(By.Id("username")));
}

public void TestLogin()
{
    var usernameField = driver.FindElement(By.Id("username"));
    var passwordField = driver.FindElement(By.Id("password"));
    var loginButton = driver.FindElement(By.Id("loginButton"));

    // Validate username input is enabled and clear before sending keys
    Assert.IsTrue(usernameField.Enabled, "Username field is not enabled.");
    usernameField.Clear();
    usernameField.SendKeys("testuser");

    // Validate password input similarly
    Assert.IsTrue(passwordField.Enabled, "Password field is not enabled.");
    passwordField.Clear();
    passwordField.SendKeys("password123");

    loginButton.Click();

    // Wait for either welcome message or error message to appear
    WebDriverWait wait = new WebDriverWait(driver, TimeSpan.FromSeconds(10));
    bool loginSuccess = false;
    try
    {
        wait.Until(SeleniumExtras.WaitHelpers.ExpectedConditions.ElementIsVisible(By.Id("welcomeMessage")));
        loginSuccess = true;
    }
    catch (WebDriverTimeoutException)
    {
        // If welcome message not found, check for error message
        var errorMessage = driver.FindElement(By.Id("loginError"));
        Assert.IsTrue(errorMessage.Displayed, "Login failed but no error message displayed.");
        Assert.IsTrue(errorMessage.Text.Contains("Invalid username"), "Error message does not indicate username issue.");
    }

    Assert.IsTrue(loginSuccess, "Login was not successful.");
}


Changes Made:  
- Added explicit waits in Setup to ensure the username field is visible before interaction, preventing timing issues.  
- Added assertions to verify the username and password fields are enabled and cleared before input, ensuring proper input handling.  
- Enhanced TestLogin to handle both successful login (welcome message) and failed login scenarios (error message), specifically checking for username-related error messages.  
- Added error handling with try-catch to differentiate between success and failure states and assert accordingly.  
- Included validation of error messages to confirm the username functionality is correctly validated and errors surfaced.
