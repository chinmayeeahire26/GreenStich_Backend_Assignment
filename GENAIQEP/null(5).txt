import org.openqa.selenium.support.ui.ExpectedConditions;
import org.openqa.selenium.support.ui.WebDriverWait;
import java.time.Duration;
import org.openqa.selenium.TimeoutException;
import org.openqa.selenium.NoSuchElementException;

public class LoginTest {
    WebDriver driver;
    WebDriverWait wait;

    // Credentials parameterized
    private String username = System.getProperty("login.username", "testuser");
    private String password = System.getProperty("login.password", "password");

    @Before
    public void setUp() {
        // Use configurable chromedriver path with fallback
        String chromeDriverPath = System.getProperty("webdriver.chrome.driver");
        if (chromeDriverPath == null || chromeDriverPath.isEmpty()) {
            throw new RuntimeException("System property 'webdriver.chrome.driver' is not set. Please set it to the path of chromedriver executable.");
        }
        System.setProperty("webdriver.chrome.driver", chromeDriverPath);

        driver = new ChromeDriver();
        driver.manage().window().maximize();
        driver.get("https://example.com/login");

        wait = new WebDriverWait(driver, Duration.ofSeconds(10));
    }

    public void loginApplication() {
        try {
            // Wait until username field is visible and enabled before interacting
            WebElement usernameField = wait.until(ExpectedConditions.elementToBeClickable(By.id("username")));
            usernameField.clear();
            usernameField.sendKeys(username); // parameterized username

            WebElement passwordField = wait.until(ExpectedConditions.elementToBeClickable(By.id("password")));
            passwordField.clear();
            passwordField.sendKeys(password); // parameterized password

            WebElement loginBtn = wait.until(ExpectedConditions.elementToBeClickable(By.id("loginBtn")));
            loginBtn.click();

            // Wait for logout button presence and visibility to confirm login success
            WebElement logoutBtn = wait.until(ExpectedConditions.visibilityOfElementLocated(By.id("logoutBtn")));

            // Additional verification: check for user profile element or welcome message
            WebElement userProfile = wait.until(ExpectedConditions.visibilityOfElementLocated(By.id("userProfile")));
            Assert.assertTrue(logoutBtn.isDisplayed() && userProfile.isDisplayed());

        } catch (TimeoutException e) {
            throw new RuntimeException("Login failed due to element not found or not interactable within timeout.", e);
        } catch (NoSuchElementException e) {
            throw new RuntimeException("Login failed due to missing element.", e);
        }
    }
}


Changes Made:  
- Added explicit waits before interacting with username, password, and login button elements to ensure they are ready.  
- Added exception handling (TimeoutException, NoSuchElementException) to provide clear diagnostics on failures.  
- Parameterized credentials using system properties (`login.username` and `login.password`) instead of hardcoded values.  
- Added additional login success verification by checking visibility of a user profile element alongside the logout button.  
- Replaced hardcoded chromedriver path with a required system property check, throwing a runtime exception if missing.  
- Added `clear()` calls before sending keys to input fields for reliability.  
- Maximized browser window on startup for consistent UI element interaction.
