    HttpRequest request = HttpRequest.newBuilder()

        .header("Accept", "application/json")
        .GET()
        .build();

    HttpResponse<String> response = httpClient.send(request, HttpResponse.BodyHandlers.ofString());

    if (response.statusCode() == 404) {
        throw new ApiException("Fetch testcases API endpoint not found: " + endpoint);

        throw new ApiException("Failed to fetch testcases: HTTP " + response.statusCode());
    }

    // Parse JSON response to list of Testcase objects
