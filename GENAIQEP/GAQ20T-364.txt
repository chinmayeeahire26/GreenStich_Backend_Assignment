// DriverFactory.cs (UPDATED) - stabilize hover/popup behavior by forcing a consistent window size in ALL modes

using OpenQA.Selenium;
using OpenQA.Selenium.Chrome;

public static class DriverFactory
{
    public static IWebDriver Create(bool headless = false)
    {
        var options = new ChromeOptions();

        if (headless)
        {
            options.AddArgument("--headless=new");
        }

        // ✅ FIX: consistent viewport for tooltip/popup positioning and hover targets (headed + headless)
        options.AddArgument("--window-size=1920,1080"); // NEW (previously only set in headless)

        // (Optional but recommended for UI stability)
        options.AddArgument("--disable-gpu");
        options.AddArgument("--no-sandbox");

        return new ChromeDriver(options);
    }
}

// UiWaits.cs (NEW) - reusable helpers to reduce flakiness and enforce a standard tooltip/popup validation pattern

using OpenQA.Selenium;
using OpenQA.Selenium.Support.UI;
using SeleniumExtras.WaitHelpers;

public static class UiWaits
{
    public static IWebElement WaitVisible(WebDriverWait wait, By locator) =>
        wait.Until(ExpectedConditions.ElementIsVisible(locator));

    public static IWebElement WaitClickable(WebDriverWait wait, By locator) =>
        wait.Until(ExpectedConditions.ElementToBeClickable(locator));
}

using OpenQA.Selenium;
using OpenQA.Selenium.Interactions;
using OpenQA.Selenium.Support.UI;

public class ContainerPage
{
    private readonly IWebDriver _driver;
    private readonly WebDriverWait _wait;

    public ContainerPage(IWebDriver driver, WebDriverWait wait)
    {
        _driver = driver;
        _wait = wait;
    }

    // ✅ RECOMMENDED: ask devs to add these attributes in the app for stable automation:
    // data-testid="container-name-input"
    // data-testid="container-name-tooltip"
    // data-testid="container-name-error"
    private By ContainerNameInput => By.CssSelector("[data-testid='container-name-input']");
    private By ContainerNameTooltip => By.CssSelector("[data-testid='container-name-tooltip']");
    private By ContainerNameErrorPopup => By.CssSelector("[data-testid='container-name-error']");

    public void HoverContainerName()
    {
        var input = UiWaits.WaitVisible(_wait, ContainerNameInput);
        new Actions(_driver).MoveToElement(input).Perform(); // ✅ hover triggers tooltip
    }

    public string GetTooltipText()
    {
        var tooltip = UiWaits.WaitVisible(_wait, ContainerNameTooltip);
        return tooltip.Text?.Trim() ?? "";
    }

    public void TypeContainerName(string name)
    {
        var input = UiWaits.WaitVisible(_wait, ContainerNameInput);
        input.Clear();
        input.SendKeys(name);
    }

    public void BlurContainerName()
    {
        // ✅ triggers validation in many UIs (blur/tab out)
        var input = UiWaits.WaitVisible(_wait, ContainerNameInput);
        input.SendKeys(Keys.Tab);
    }

    public string GetValidationErrorText()
    {
        var error = UiWaits.WaitVisible(_wait, ContainerNameErrorPopup);
        return error.Text?.Trim() ?? "";
    }
}

using NUnit.Framework;
using OpenQA.Selenium;
using OpenQA.Selenium.Support.UI;
using System;

[TestFixture]

{
    private IWebDriver driver = null!;
    private WebDriverWait wait = null!;
    private TestSettings settings = null!;

    [SetUp]
    public void Setup()
    {
        settings = Config.Load(); // existing config loader, now includes ContainerNaming
        driver = DriverFactory.Create(headless: false);
        wait = new WebDriverWait(driver, TimeSpan.FromSeconds(10));
    }

    [TearDown]
    public void TearDown()
    {
        driver.Quit();
    }

    [Test]
    public void ContainerName_Hover_ShowsNamingConventionTooltip()

        // ✅ NOTE: Update navigation to the real page where container name input exists
        driver.Navigate().GoToUrl($"{settings.BaseUrl}/containers/create"); // NEW (example route)

        var page = new ContainerPage(driver, wait);

        page.HoverContainerName();
        var tooltipText = page.GetTooltipText();

        // ✅ Enforces RCA: required guidance text must be present in hover help
        Assert.That(tooltipText, Does.Contain(settings.ContainerNaming.ConventionText));

    [Test]
    public void ContainerName_InvalidValue_ShowsErrorPopupWithNamingConvention()
