public void loginToCapgemini(WebDriver driver) {
    try {
        driver.get("https://flipkart.com/login");

        // Fetch credentials securely from environment variables or config
        String username = System.getenv("FLIPKART_USERNAME");
        String password = System.getenv("FLIPKART_PASSWORD");
        if (username == null || password == null) {
            throw new IllegalStateException("Login credentials not set in environment variables");
        }

        WebElement usernameField = driver.findElement(By.id("username"));
        WebElement passwordField = driver.findElement(By.id("password"));
        usernameField.clear();
        usernameField.sendKeys(username);
        passwordField.clear();
        passwordField.sendKeys(password);

        // Corrected locator for login button: use By.name("login") instead of By.id("loginButton")
        WebElement loginButton = driver.findElement(By.name("login")); // Fixed locator
        loginButton.click();

        // Wait for page load or dashboard elements to appear
        WebDriverWait wait = new WebDriverWait(driver, Duration.ofSeconds(10));
        // Check for presence of welcome banner or logout link as login success indicators
        boolean loginSuccess = false;
        try {
            wait.until(ExpectedConditions.or(
                ExpectedConditions.visibilityOfElementLocated(By.className("welcome")),
                ExpectedConditions.visibilityOfElementLocated(By.cssSelector("a[href='/logout']"))
            ));
            loginSuccess = true;
        } catch (TimeoutException te) {
            loginSuccess = false;
        }

        if (loginSuccess) {
            System.out.println("Login successful");
        } else {
            System.out.println("Login failed: Dashboard elements not found");
            // Optionally, capture screenshot or log more details here
        }
    } catch (IllegalStateException ise) {
        System.err.println("Configuration error: " + ise.getMessage());
    } catch (NoSuchElementException nse) {
        System.err.println("Element not found during login process: " + nse.getMessage());
    } catch (Exception e) {
        System.err.println("Unexpected error during login: " + e.getMessage());
        e.printStackTrace();
    }
}


Changes Made:  
- Changed login button locator from `By.id("loginButton")` to `By.name("login")` to match actual page element.  
- Updated login success verification to check for elements with class `"welcome"` or logout link with href `"/logout"` instead of non-existent `id="dashboard"`.  
- Replaced hardcoded credentials with environment variable retrieval (`FLIPKART_USERNAME` and `FLIPKART_PASSWORD`) and added validation for their presence.  
- Added explicit waits (`WebDriverWait`) to handle asynchronous page loading and element visibility.  
- Enhanced error handling with specific catch blocks for configuration errors, element not found, and general exceptions, providing meaningful error messages.  
- Added clearing of input fields before sending keys to avoid input issues.
