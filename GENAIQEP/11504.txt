public class LoginTest {
    WebDriver driver;
    String baseUrl;
    String environment;

    @Before
    public void setUp() {
        // Environment parameterization for flexibility
        environment = System.getProperty("env", "old-prod"); // default to old-prod if not set

        // Set baseUrl and driver path based on environment
        switch (environment) {
            case "new-prod":
                baseUrl = "https://new-prod.example.com";
                System.setProperty("webdriver.chrome.driver", "/path/to/new-prod/chromedriver");
                break;
            case "test":
                baseUrl = "https://test.example.com";
                System.setProperty("webdriver.chrome.driver", "/path/to/test/chromedriver");
                break;
            default:
                baseUrl = "https://old-prod.example.com";
                System.setProperty("webdriver.chrome.driver", "/path/to/old-prod/chromedriver");
        }

        driver = new ChromeDriver();
        driver.get(baseUrl + "/login");
    }

    @Test
    public void verifyLoginFunctionality() {
        driver.findElement(By.id("username")).sendKeys("user");
        driver.findElement(By.id("password")).sendKeys("pass");
        driver.findElement(By.id("loginBtn")).click();
        Assert.assertTrue(driver.findElement(By.id("welcomeMsg")).isDisplayed());
    }

    @Test
    public void verifyHelpSectionContentAndUI() {
        // Navigate to Help section after login
        verifyLoginFunctionality(); // Reuse login steps

        driver.findElement(By.id("helpMenu")).click();

        // Wait for Help content to load (explicit wait recommended)
        WebDriverWait wait = new WebDriverWait(driver, Duration.ofSeconds(10));
        WebElement helpContent = wait.until(ExpectedConditions.visibilityOfElementLocated(By.id("helpContent")));

        // Verify Help content is updated as expected for the environment
        String expectedHelpText = getExpectedHelpTextForEnvironment(environment);
        String actualHelpText = helpContent.getText();
        Assert.assertTrue(actualHelpText.contains(expectedHelpText), "Help content is not updated correctly.");

        // UI validation: check content color (example)
        String contentColor = helpContent.getCssValue("color");
        String expectedColor = "rgba(0, 0, 0, 1)"; // black color expected
        Assert.assertEquals(contentColor, expectedColor, "Help content color is incorrect.");

        // Additional UI rendering checks can be added here

        // Error handling: verify no error messages displayed
        List<WebElement> errorMessages = driver.findElements(By.className("help-error"));
        Assert.assertTrue(errorMessages.isEmpty(), "Help section contains error messages.");
    }

    private String getExpectedHelpTextForEnvironment(String env) {
        switch (env) {
            case "new-prod":
                return "Updated Help Content for New Production";
            case "test":
                return "Help Content for Test Environment";
            default:
                return "Help Content for Old Production";
        }
    }

    @After
    public void tearDown() {
        if (driver != null) {
            driver.quit();
        }
    }
}
